Classes e Objetos
Anatomia de uma classe bem escrita
public class ContaBancaria {
    private String titular;      // private = encapsulamento
    private double saldo;
    private static int total = 0; // estático: compartilhado pela classe

    public ContaBancaria(String t, double s) {
        this.titular = t;  this.saldo = s;  total++;
    }
    public ContaBancaria() { this("Sem nome", 0); } // delega

    public String  getTitular()  { return titular; }
    public double  getSaldo()    { return saldo; }
    public static int getTotal() { return total; }

    public void   depositar(double v) { if(v>0) saldo+=v; }
    public boolean sacar(double v) {
        if(v>0 && saldo>=v) { saldo-=v; return true; }
        return false;
    }
    @Override public String toString() {
        return "Conta{"+titular+", R$"+saldo+"}";
    }
}
ContaBancaria c = new ContaBancaria("Rafael", 1000);
c.depositar(500);  System.out.println(c); // chama toString()

Herança e Polimorfismo
extends + @Override + super
public class Veiculo {
    protected String marca;
    public Veiculo(String m) { marca = m; }
    public String desc() { return "Veículo: "+marca; }
}
public class Carro extends Veiculo {
    private int portas;
    public Carro(String m, int p) { super(m); portas=p; } // super() primeiro!
    @Override public String desc() {
        return super.desc()+" | Portas: "+portas;
    }
}
// Polimorfismo: referência pai, objeto filho
Veiculo v = new Carro("Toyota", 4);
v.desc(); // chama método do Carro (binding dinâmico)

// instanceof + pattern matching (Java 16+)
if (v instanceof Carro c) { System.out.println(c.portas); }
Abstract e Interface
abstract class vs interface
public abstract class Forma {
    protected String cor = "preto";             // tem atributo
    public abstract double area();               // força implementação
    public void exibir() { System.out.println(area()); }
}
public interface Desenhavel {
    void desenhar();                              // implicitamente public abstract
    default void zoom(double f) { System.out.println("x"+f); } // Java 8+
    static String versao() { return "1.0"; }
}
// extends UMA class + implements VÁRIAS interfaces
public class Circulo extends Forma implements Desenhavel {
    private double r;
    public Circulo(double r) { this.r=r; }
    @Override public double area() { return Math.PI*r*r; }
    @Override public void   desenhar() { System.out.println("O"); }
}

try / catch / finally
Sintaxe completa
try {
    int r = 10 / 0;

} catch (ArithmeticException e) {        // específico PRIMEIRO
    System.out.println("Divisão por zero: " + e.getMessage());

} catch (NullPointerException | ArrayIndexOutOfBoundsException e) {
    e.printStackTrace();                   // multi-catch (Java 7+)

} catch (Exception e) {                  // genérico por ÚLTIMO
    System.out.println("Erro: " + e.getMessage());

} finally {
    System.out.println("SEMPRE executa"); // use para fechar recursos
}

// try-with-resources (Java 7+) — fecha automaticamente
try (Scanner sc = new Scanner(new File("arq.txt"))) {
    while (sc.hasNextLine()) System.out.println(sc.nextLine());
} // sc.close() chamado automaticamente!
⚠ Ordem dos catches importa!Sempre coloque catches específicos ANTES dos genéricos. Exception antes de IOException = erro de compilação.
throw, throws e exceção customizada
// throws — declara que PODE lançar checked exception
public void ler() throws IOException { ... }

// throw — lança explicitamente
public void setIdade(int i) {
    if (i < 0) throw new IllegalArgumentException("Negativo: "+i);
    this.idade = i;
}

// Exceção customizada
public class SaldoInsuficienteException extends RuntimeException {
    private final double saldo;
    public SaldoInsuficienteException(double s) {
        super("Saldo insuficiente: R$"+s);
        this.saldo = s;
    }
    public double getSaldo() { return saldo; }
}

String — imutável

String s = "  Hello, World!  ";

s.length();                    // 17

s.indexOf("o");               // primeira ocorrência
s.lastIndexOf("o");           // última

s.toLowerCase();               // tudo minúsculo
s.toUpperCase();               // tudo maiúsculo
s.replace("World","Java");   // substitui todas ocorrências

String.format("%s=%d","n",5); // "n=5"
String.join("-","a","b","c");// "a-b-c"
⚠ String é imutável!Qualquer operação retorna um NOVO objeto. Você DEVE atribuir: s = s.trim();.

// Classe genérica
public class Par<A, B> {
    private final A p; private final B s;
    public Par(A a, B b) { p=a; s=b; }
    public A getPrimeiro() { return p; }
    public B getSegundo()  { return s; }
}
Par<String,Integer> par = new Par<>("Java", 21);

// Método genérico com bound
public static <T extends Comparable<T>> T max(T a, T b) {
    return a.compareTo(b) > 0 ? a : b;
}

// Wildcards
void printAll(List<?> l)                  // qualquer tipo
void addNums(List<? super Integer> l)    // Integer ou superclasse
void readNums(List<? extends Number> l)  // Number ou subclasse
Lambdas e Interfaces Funcionais
As 4 interfaces core do java.util.function
// Lambda: (params) -> expressão  ou  (params) -> { bloco }

// Predicate<T>  — T -> boolean
Predicate<Integer> ehPar = n -> n % 2 == 0;
ehPar.test(4);    // true
ehPar.and(n -> n > 0); // encadeamento

// Function<T,R> — T -> R
Function<String,Integer> tam = String::length; // method ref
tam.apply("Java");  // 4
tam.andThen(n -> n * 2).apply("Java"); // 8

// Consumer<T>  — T -> void
Consumer<String> log = s -> System.out.println("[LOG] "+s);
log.accept("ok");

// Supplier<T>  — () -> T
Supplier<String> saud = () -> "Olá!";
saud.get();

// Method references: Classe::método ou obj::método
List.of("a","b").forEach(System.out::println);
Stream API
Pipeline funcional sobre coleções
List<Integer> nums = List.of(3,1,4,1,5,9,2,6);

// filter + map + collect
List<String> r = nums.stream()
    .filter(n -> n > 3)
    .map(n -> "#"+n)
    .sorted()
    .collect(Collectors.toList());

// Terminais úteis
long    qtd  = nums.stream().filter(n->n%2==0).count();
int     soma = nums.stream().mapToInt(Integer::intValue).sum();
boolean algum = nums.stream().anyMatch(n->n%2==0);
boolean todos = nums.stream().allMatch(n->n>0);
boolean nenhum= nums.stream().noneMatch(n->n<0);
Optional<Integer> max = nums.stream().max(Integer::compareTo);

// distinct, limit, skip
nums.stream().distinct().limit(3).skip(1).toList();

// Collectors
String joined = nums.stream().map(Object::toString)
    .collect(Collectors.joining(", "));
💡 Streams são lazy e não modificam a coleção original!Operações intermediárias (filter, map) só executam quando o terminal (collect, count, forEach) é chamado.
Optional — evita NullPointerException
Optional<String> p = Optional.of("valor");      // NPE se null
Optional<String> v = Optional.empty();
Optional<String> s = Optional.ofNullable(x);  // seguro

p.isPresent();                  // true
p.isEmpty();                    // false (Java 11)
p.get();                         // "valor" — lança se vazio!
p.orElse("padrão");            // valor ou "padrão"
p.orElseGet(() -> calc());      // lazy
p.orElseThrow();                 // NoSuchElementException
p.map(String::toUpperCase);    // transforma sem desempacotar
p.filter(x -> x.length()>3); // filtra
p.ifPresent(System.out::println);
